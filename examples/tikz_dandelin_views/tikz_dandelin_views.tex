% One authoritative ellipse construction shown in three distinct static views.
% The painter policy is authored here rather than left to Provider defaults:
% the spatial view certifies camera-dependent cone/sphere/plane layering, while
% the two planar reductions use their explicit diagrammatic draw contracts.
% In the spatial view, plane regions behind either near cone sheet use a muted
% fill and their finite outline fragments become dashed automatically.

% 1. Fixed spatial teaching view.  The contact circles remain visible because
% they own the equal-depth seams between each sphere and its tangent cone sheet.
\begin{tikzpicture}[3d view={38}{24},scale=0.58]
  \coordinate (A) at (0,0,0);
  \coordinate (Z) at (0,0,1);
  \coordinate (R) at (1,0,0);
  \coordinate (O) at (0,0,2);
  \coordinate (U) at (0,1,2);
  \coordinate (V) at (-0.8,0,2.6);
  \DeclareSpacePlane{cut}{O/U/V};
  \DeclareSpaceRightCone{cone}{A/Z/R}{30}{0/9}{open_single};
  \DeclareDandelinConstruction{dan}{cone}{cut};
  \DrawDandelinDiagram[
    view=spatial,
    mode=depth_aware_teaching_transparent,
    preset=classroom,
    show-contact-circles=true,
    show-foci=true,
    show-directrices=false
  ]{dan};
\end{tikzpicture}
